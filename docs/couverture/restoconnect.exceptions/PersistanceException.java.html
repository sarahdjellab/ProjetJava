package restoconnect.exceptions;
/** Encapsule un échec technique de stockage sans exposer une erreur SQL à la vue. */
public class PersistanceException extends RuntimeException {
    /**
     * @param message explication métier
     * @param cause erreur technique conservée pour le journal */
    public PersistanceException(String message, Throwable cause) { super(message, cause); }
}

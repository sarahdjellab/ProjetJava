package restoconnect.modele;

import jakarta.persistence.*;

/**
 * Représente une table du restaurant.
 */
@Entity
@jakarta.persistence.Table(name = "tables_restaurant")
public class Table {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    /** Constructeur réservé à Hibernate. */
    protected Table() { }

    /**
     * @return identifiant technique, vide avant la première sauvegarde */
    public java.util.Optional<Long> getId() {
        return java.util.Optional.ofNullable(id);
    }


    @Column(nullable = false, unique = true)
    private int numero;
    @Column(nullable = false)
    private int capacite;

    /**
     * Construit une table.
     *
     *
     * @param numero numéro de la table
     *
     * @param capacite nombre maximal de personnes
     *
     *
     * @throws IllegalArgumentException si la capacité est
     *                                  inférieure ou égale à zéro
     */
    public Table(int numero, int capacite) {

        if (numero <= 0) {
            throw new IllegalArgumentException(
                    "Le numéro de table doit être positif."
            );
        }

        if (capacite <= 0) {
            throw new IllegalArgumentException(
                    "La capacité doit être strictement positive."
            );
        }

        this.numero = numero;
        this.capacite = capacite;
    }

    /**
     * Retourne le numéro de la table.
     *
     *
     * @return numéro
     */
    public int getNumero() {
        return numero;
    }

    /**
     * Modifie le numéro de la table.
     *
     *
     * @param numero nouveau numéro
     */
    public void setNumero(int numero) {

        if (numero <= 0) {
            throw new IllegalArgumentException(
                    "Le numéro doit être positif."
            );
        }

        this.numero = numero;
    }

    /**
     * Retourne la capacité de la table.
     *
     *
     * @return capacité
     */
    public int getCapacite() {
        return capacite;
    }

    /**
     * Modifie la capacité de la table.
     *
     *
     * @param capacite nouvelle capacité
     */
    public void setCapacite(int capacite) {

        if (capacite <= 0) {
            throw new IllegalArgumentException(
                    "La capacité doit être strictement positive."
            );
        }

        this.capacite = capacite;
    }

    /**
     * Retourne une représentation textuelle de la table.
     *
     *
     * @return informations de la table
     */
    @Override
    public String toString() {
        return "Table " + numero +
                " (" + capacite + " personnes)";
    }
}

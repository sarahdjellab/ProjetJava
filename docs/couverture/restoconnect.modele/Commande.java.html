package restoconnect.modele;

import jakarta.persistence.*;

import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

/**
 * Représente une commande prise en salle.
 *
 * Une commande est liée à une réservation et à un serveur.
 * Elle contient une ou plusieurs lignes de commande.
 */
@Entity
public class Commande {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    /** Constructeur réservé à Hibernate. */
    protected Commande() { }

    /**
     * @return identifiant technique, vide avant la première sauvegarde */
    public java.util.Optional<Long> getId() {
        return java.util.Optional.ofNullable(id);
    }


    @OneToOne(optional = false, fetch = FetchType.EAGER)
    private Reservation reservation;
    // Association : le serveur prend en charge la commande.
    @ManyToOne(optional = false, fetch = FetchType.EAGER)
    private Serveur serveur;
    @ElementCollection(fetch = FetchType.EAGER)
    @CollectionTable(name = "lignes_commande", joinColumns = @JoinColumn(name = "commande_id"))
    @OrderColumn(name = "position_ligne")
    private List<LigneCommande> ligneCommandes = new ArrayList<>();

    @Column(nullable = false)
    private boolean cloturee;
    @Column(nullable = false)
    private double remise;
    @Column(nullable = false)
    private double montantFacture;

    /**
     * Construit une commande avec une première ligne.
     *
     * Le fait de demander une première ligne dans le constructeur
     * permet de respecter l'invariant du sujet :
     * une commande doit contenir au moins une ligne.
     *
     *
     * @param reservation réservation associée
     *
     * @param serveur serveur responsable
     *
     * @param article premier article commandé
     *
     * @param quantite quantité du premier article
     *
     *
     * @throws IllegalArgumentException si un paramètre est invalide
     */
    public Commande(
            Reservation reservation,
            Serveur serveur,
            ArticleMenu article,
            int quantite) {

        if (reservation == null) {
            throw new IllegalArgumentException(
                    "La réservation est obligatoire."
            );
        }

        if (serveur == null) {
            throw new IllegalArgumentException(
                    "Le serveur est obligatoire."
            );
        }

        if (article == null) {
            throw new IllegalArgumentException(
                    "L'article est obligatoire."
            );
        }

        if (quantite <= 0) {
            throw new IllegalArgumentException(
                    "La quantité doit être strictement positive."
            );
        }

        this.reservation = reservation;
        this.serveur = serveur;
        this.ligneCommandes = new ArrayList<>();

        // Première ligne obligatoire.
        this.ligneCommandes.add(
                new LigneCommande(article, quantite)
        );
    }

    /**
     * Retourne la réservation associée.
     *
     *
     * @return réservation
     */
    public Reservation getReservation() {
        return reservation;
    }

    /**
     * Retourne le serveur associé.
     *
     *
     * @return serveur
     */
    public Serveur getServeur() {
        return serveur;
    }

    /**
     * Retourne les lignes de la commande.
     *
     * Une vue non modifiable est retournée afin d'éviter
     * qu'un code extérieur puisse casser l'encapsulation.
     *
     *
     * @return liste des lignes
     */
    public List<LigneCommande> getLigneCommandes() {
        return Collections.unmodifiableList(ligneCommandes);
    }

    /**
     * Ajoute une ligne à la commande.
     *
     *
     * @param article article commandé
     *
     * @param quantite quantité
     *
     *
     * @throws IllegalArgumentException si l'article est null
     *                                  ou si la quantité est invalide
     */
    public void ajouterLigne(ArticleMenu article, int quantite) {

        if (article == null) {
            throw new IllegalArgumentException(
                    "L'article est obligatoire."
            );
        }

        if (quantite <= 0) {
            throw new IllegalArgumentException(
                    "La quantité doit être strictement positive."
            );
        }

        if (cloturee) {
            throw new IllegalStateException("Une commande clôturée ne peut plus être modifiée.");
        }
        ligneCommandes.add(
                new LigneCommande(article, quantite)
        );
    }

    /**
     * Calcule le montant total de la commande.
     *
     *
     * @return montant total
     */
    public double calculerTotal() {

        double total = 0;

        for (LigneCommande ligne : ligneCommandes) {
            total += ligne.calculerMontant();
        }

        // Invariant interne : seules des lignes validées ont été ajoutées.
        // Une assertion signale ici un défaut du programme, pas une saisie invalide.
        assert !ligneCommandes.isEmpty() : "Une commande doit contenir une ligne.";
        return cloturee ? montantFacture : total * (1 - remise);
    }

    /**
     * @return vrai si la facture a été figée */
    public boolean isCloturee() { return cloturee; }

    /**
     * Fixe la remise avant facturation.
     *
     * @param remise taux compris entre zéro et un exclu
     *
     * @throws IllegalArgumentException si le taux est invalide
     *
     * @throws IllegalStateException si la commande est clôturée
     */
    public void setRemise(double remise) {
        if (!Double.isFinite(remise) || remise < 0 || remise >= 1) {
            throw new IllegalArgumentException("La remise doit être comprise entre 0 et 1 exclu.");
        }
        if (cloturee) { throw new IllegalStateException("La commande est déjà clôturée."); }
        this.remise = remise;
    }

    /**
     * Fige le montant de la facture.
     * Précondition : commande ouverte et non vide. Postcondition : montant stable.
     *
     * @throws IllegalStateException si la commande est déjà clôturée ou vide
     */
    public void cloturer() {
        if (cloturee) { throw new IllegalStateException("La commande est déjà clôturée."); }
        if (ligneCommandes.isEmpty()) { throw new IllegalStateException("Une commande vide ne peut pas être clôturée."); }
        montantFacture = calculerTotal();
        cloturee = true;
    }

    /**
     * Classe interne représentant une ligne de commande.
     *
     * Cette classe est volontairement imbriquée dans Commande.
     * Elle est statique pour permettre son instanciation par Hibernate ;
     * son constructeur métier reste privé et son cycle de vie dépend de Commande.
     * Une LigneCommande n'a aucun sens sans la Commande
     * qui la contient.
     *
     * Cela représente une relation de composition.
     */
    @Embeddable
    public static class LigneCommande {

        @ManyToOne(optional = false, fetch = FetchType.EAGER)
        private ArticleMenu articleMenu;
        @Column(nullable = false)
        private double prixUnitaireTTC;

        /** Constructeur réservé à Hibernate. */
        protected LigneCommande() { }
        @Column(nullable = false)
        private int quantite;

        /**
         * Construit une ligne de commande.
         *
         * Le constructeur est privé car une ligne doit être
         * créée uniquement depuis une Commande.
         *
         *
     * @param articleMenu article commandé
         *
     * @param quantite quantité
         */
        private LigneCommande(
                ArticleMenu articleMenu,
                int quantite) {

            if (articleMenu == null) {
                throw new IllegalArgumentException(
                        "L'article est obligatoire."
                );
            }

            if (quantite <= 0) {
                throw new IllegalArgumentException(
                        "La quantité doit être positive."
                );
            }

            this.articleMenu = articleMenu;
            this.prixUnitaireTTC = articleMenu.calculerPrixTTC();
            this.quantite = quantite;
        }

        /**
         * Retourne l'article.
         *
         *
     * @return article
         */
        public ArticleMenu getArticleMenu() {
            return articleMenu;
        }

        /**
         * Retourne la quantité.
         *
         *
     * @return quantité
         */
        public int getQuantite() {
            return quantite;
        }

        /**
         * Calcule le montant de cette ligne.
         *
         *
     * @return prix unitaire multiplié par la quantité
         */
        public double calculerMontant() {
            return prixUnitaireTTC * quantite;
        }
    }
}

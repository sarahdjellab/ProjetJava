package restoconnect.modele;
import jakarta.persistence.*;
import java.util.*;
/** Article de catégorie Dessert ; coefficient pédagogique de 1.2 (pas un taux fiscal réel). */
@Entity
@DiscriminatorValue("DESSERT")
public class Dessert extends ArticleMenu  {
    /** Constructeur réservé à Hibernate. */
    protected Dessert() { }
    /**
     * @param nom nom non vide
     * @param prixBase prix hors majoration, strictement positif */
    public Dessert(String nom, double prixBase) { super(nom, prixBase); }
    /**
     * @param nom nom non vide ; prix par défaut de dix euros */
    public Dessert(String nom) { this(nom, 10.0); }
    /** {@inheritDoc} */
    @Override
    public double calculerPrixTTC() { return (getPrixBase()) * 1.2; }
}

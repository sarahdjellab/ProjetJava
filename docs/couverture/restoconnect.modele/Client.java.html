package restoconnect.modele;

import jakarta.persistence.*;

/**
 * Représente un client du restaurant.
 *
 * Un client possède un nom, un numéro de téléphone
 * et une adresse e-mail.
 */
@Entity
public class Client {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    /** Constructeur réservé à Hibernate. */
    protected Client() { }

    /**
     * @return identifiant technique, vide avant la première sauvegarde */
    public java.util.Optional<Long> getId() {
        return java.util.Optional.ofNullable(id);
    }


    @Column(nullable = false)
    private String nom;
    @Column(nullable = false)
    private String telephone;
    @Column(nullable = false)
    private String email;

    /**
     * Construit un client.
     *
     *
     * @param nom nom du client
     *
     * @param telephone numéro de téléphone
     *
     * @param email adresse e-mail
     *
     *
     * @throws IllegalArgumentException si le nom est vide
     *                                  ou si l'e-mail ne contient pas @
     */
    public Client(String nom, String telephone, String email) {

        if (nom == null || nom.isBlank()) {
            throw new IllegalArgumentException(
                    "Le nom du client ne peut pas être vide."
            );
        }

        if (email == null || !email.contains("@")) {
            throw new IllegalArgumentException(
                    "L'adresse e-mail doit contenir un @."
            );
        }

        this.nom = nom;
        this.telephone = telephone == null ? "" : telephone;
        this.email = email;
    }

    /**
     * Retourne le nom du client.
     *
     *
     * @return nom du client
     */
    public String getNom() {
        return nom;
    }

    /**
     * Modifie le nom du client.
     *
     *
     * @param nom nouveau nom
     */
    public void setNom(String nom) {

        if (nom == null || nom.isBlank()) {
            throw new IllegalArgumentException(
                    "Le nom du client ne peut pas être vide."
            );
        }

        this.nom = nom;
    }

    /**
     * Retourne le téléphone du client.
     *
     *
     * @return téléphone
     */
    public String getTelephone() {
        return telephone;
    }

    /**
     * Modifie le téléphone du client.
     *
     *
     * @param telephone nouveau téléphone
     */
    public void setTelephone(String telephone) {
        this.telephone = telephone == null ? "" : telephone;
    }

    /**
     * Retourne l'e-mail du client.
     *
     *
     * @return adresse e-mail
     */
    public String getEmail() {
        return email;
    }

    /**
     * Modifie l'e-mail du client.
     *
     *
     * @param email nouvel e-mail
     */
    public void setEmail(String email) {

        if (email == null || !email.contains("@")) {
            throw new IllegalArgumentException(
                    "L'adresse e-mail doit contenir un @."
            );
        }

        this.email = email;
    }

    /**
     * Retourne une représentation textuelle du client.
     *
     *
     * @return informations principales du client
     */
    @Override
    public String toString() {
        return "Client{" +
                "nom='" + nom + '\'' +
                ", telephone='" + telephone + '\'' +
                ", email='" + email + '\'' +
                '}';
    }
}

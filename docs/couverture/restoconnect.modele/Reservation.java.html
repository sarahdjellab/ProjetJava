package restoconnect.modele;

import jakarta.persistence.*;

import java.time.LocalDateTime;

/**
 * Représente une réservation dans le restaurant.
 *
 * Une réservation est liée à un client et à une table.
 */
@Entity
public class Reservation {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    /** Constructeur réservé à Hibernate. */
    protected Reservation() { }

    /**
     * @return identifiant technique, vide avant la première sauvegarde */
    public java.util.Optional<Long> getId() {
        return java.util.Optional.ofNullable(id);
    }


    @Column(nullable = false)
    private LocalDateTime dateHeure;
    @Column(nullable = false)
    private int nombrePersonnes;
    // Agrégation : le client existe indépendamment de cette réservation.
    @ManyToOne(optional = false, fetch = FetchType.EAGER)
    private Client client;
    // Agrégation : une réservation ne possède pas la table.
    @ManyToOne(optional = false, fetch = FetchType.EAGER)
    private Table table;

    /**
     * Construit une réservation.
     *
     *
     * @param dateHeure date et heure de la réservation
     *
     * @param nombrePersonnes nombre de personnes
     *
     * @param client client qui réserve
     *
     * @param table table réservée
     *
     *
     * @throws IllegalArgumentException si un paramètre est invalide
     *                                  ou si le nombre de personnes
     *                                  dépasse la capacité de la table
     */
    public Reservation(
            LocalDateTime dateHeure,
            int nombrePersonnes,
            Client client,
            Table table) {

        if (dateHeure == null) {
            throw new IllegalArgumentException(
                    "La date et l'heure sont obligatoires."
            );
        }

        if (nombrePersonnes <= 0) {
            throw new IllegalArgumentException(
                    "Le nombre de personnes doit être positif."
            );
        }

        if (client == null) {
            throw new IllegalArgumentException(
                    "Le client est obligatoire."
            );
        }

        if (table == null) {
            throw new IllegalArgumentException(
                    "La table est obligatoire."
            );
        }

        if (nombrePersonnes > table.getCapacite()) {
            throw new IllegalArgumentException(
                    "Le nombre de personnes dépasse la capacité de la table."
            );
        }

        this.dateHeure = dateHeure;
        this.nombrePersonnes = nombrePersonnes;
        this.client = client;
        this.table = table;
    }

    /**
     * Retourne la date et l'heure.
     *
     *
     * @return date et heure
     */
    public LocalDateTime getDateHeure() {
        return dateHeure;
    }

    /**
     * Modifie la date et l'heure.
     *
     *
     * @param dateHeure nouvelle date
     */
    public void setDateHeure(LocalDateTime dateHeure) {

        if (dateHeure == null) {
            throw new IllegalArgumentException(
                    "La date est obligatoire."
            );
        }

        this.dateHeure = dateHeure;
    }

    /**
     * Retourne le nombre de personnes.
     *
     *
     * @return nombre de personnes
     */
    public int getNombrePersonnes() {
        return nombrePersonnes;
    }

    /**
     * Modifie le nombre de personnes.
     *
     *
     * @param nombrePersonnes nouveau nombre
     */
    public void setNombrePersonnes(int nombrePersonnes) {

        if (nombrePersonnes <= 0) {
            throw new IllegalArgumentException(
                    "Le nombre de personnes doit être positif."
            );
        }

        if (table != null && nombrePersonnes > table.getCapacite()) {
            throw new IllegalArgumentException(
                    "Le nombre de personnes dépasse la capacité."
            );
        }

        this.nombrePersonnes = nombrePersonnes;
    }

    /**
     * Retourne le client.
     *
     *
     * @return client
     */
    public Client getClient() {
        return client;
    }

    /**
     * Retourne la table.
     *
     *
     * @return table
     */
    public Table getTable() {
        return table;
    }

    /**
     * @return fin du créneau ; une réservation dure deux heures */
    public LocalDateTime getFin() {
        return dateHeure.plusHours(2);
    }

    /**
     * Retourne une représentation textuelle de la réservation.
     *
     *
     * @return informations de la réservation
     */
    @Override
    public String toString() {
        return "Reservation{" +
                "dateHeure=" + dateHeure +
                ", nombrePersonnes=" + nombrePersonnes +
                ", client=" + client.getNom() +
                ", table=" + table.getNumero() +
                '}';
    }
}

package restoconnect.modele;

import jakarta.persistence.*;

/**
 * Représente un article disponible dans le menu du restaurant.
 */
@Entity
@Inheritance(strategy = InheritanceType.SINGLE_TABLE)
@DiscriminatorColumn(name = "categorie")
public abstract class ArticleMenu {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    /** Constructeur réservé à Hibernate. */
    protected ArticleMenu() { }

    /**
     * @return identifiant technique, vide avant la première sauvegarde */
    public java.util.Optional<Long> getId() {
        return java.util.Optional.ofNullable(id);
    }


    @Column(nullable = false)
    private String nom;
    @Column(nullable = false)
    private double prixBase;

    /**
     * Construit un article avec son nom et son prix.
     *
     *
     * @param nom nom de l'article
     *
     * @param prixBase prix de base
     *
     *
     * @throws IllegalArgumentException si le nom est vide
     *                                  ou si le prix n'est pas positif
     */
    public ArticleMenu(String nom, double prixBase) {

        if (nom == null || nom.isBlank()) {
            throw new IllegalArgumentException(
                    "Le nom de l'article ne peut pas être vide."
            );
        }

        if (!Double.isFinite(prixBase) || prixBase <= 0) {
            throw new IllegalArgumentException(
                    "Le prix doit être strictement positif."
            );
        }

        this.nom = nom;
        this.prixBase = prixBase;
    }

    /**
     * Constructeur surchargé.
     *
     * Un prix par défaut de 10 euros est utilisé.
     *
     *
     * @param nom nom de l'article
     */
    public ArticleMenu(String nom) {
        this(nom, 10.0);
    }

    /**
     * Retourne le nom.
     *
     *
     * @return nom
     */
    public String getNom() {
        return nom;
    }

    /**
     * Modifie le nom.
     *
     *
     * @param nom nouveau nom
     */
    public void setNom(String nom) {

        if (nom == null || nom.isBlank()) {
            throw new IllegalArgumentException(
                    "Le nom ne peut pas être vide."
            );
        }

        this.nom = nom;
    }

    /**
     * Retourne le prix de base.
     *
     *
     * @return prix
     */
    public double getPrixBase() {
        return prixBase;
    }

    /**
     * Modifie le prix de base.
     *
     *
     * @param prixBase nouveau prix
     */
    public void setPrixBase(double prixBase) {

        if (!Double.isFinite(prixBase) || prixBase <= 0) {
            throw new IllegalArgumentException(
                    "Le prix doit être strictement positif."
            );
        }

        this.prixBase = prixBase;
    }

    /**
     * @return prix toutes taxes comprises, calculé par la catégorie */
    public abstract double calculerPrixTTC();

    /**
     * Retourne une représentation textuelle de l'article.
     *
     *
     * @return informations de l'article
     */
    @Override
    public String toString() {
        return "ArticleMenu{" +
                "nom='" + nom + '\'' +
                ", prixBase=" + prixBase +
                '}';
    }
}

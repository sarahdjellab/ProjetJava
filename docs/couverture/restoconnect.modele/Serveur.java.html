package restoconnect.modele;

import jakarta.persistence.*;

/**
 * Représente un serveur du restaurant.
 */
@Entity
public class Serveur {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    /** Constructeur réservé à Hibernate. */
    protected Serveur() { }

    /**
     * @return identifiant technique, vide avant la première sauvegarde */
    public java.util.Optional<Long> getId() {
        return java.util.Optional.ofNullable(id);
    }


    @Column(nullable = false)
    private String nom;
    @Column(nullable = false, unique = true)
    private String matricule;

    /**
     * Construit un serveur.
     *
     *
     * @param nom nom du serveur
     *
     * @param matricule matricule du serveur
     *
     *
     * @throws IllegalArgumentException si le matricule est vide
     */
    public Serveur(String nom, String matricule) {

        if (nom == null || nom.isBlank()) {
            throw new IllegalArgumentException(
                    "Le nom du serveur ne peut pas être vide."
            );
        }

        if (matricule == null || matricule.isBlank()) {
            throw new IllegalArgumentException(
                    "Le matricule ne peut pas être vide."
            );
        }

        this.nom = nom;
        this.matricule = matricule;
    }

    /**
     * Retourne le nom du serveur.
     *
     *
     * @return nom
     */
    public String getNom() {
        return nom;
    }

    /**
     * Modifie le nom du serveur.
     *
     *
     * @param nom nouveau nom
     */
    public void setNom(String nom) {

        if (nom == null || nom.isBlank()) {
            throw new IllegalArgumentException(
                    "Le nom ne peut pas être vide."
            );
        }

        this.nom = nom;
    }

    /**
     * Retourne le matricule.
     *
     *
     * @return matricule
     */
    public String getMatricule() {
        return matricule;
    }

    /**
     * Modifie le matricule.
     *
     *
     * @param matricule nouveau matricule
     */
    public void setMatricule(String matricule) {

        if (matricule == null || matricule.isBlank()) {
            throw new IllegalArgumentException(
                    "Le matricule ne peut pas être vide."
            );
        }

        this.matricule = matricule;
    }

    /**
     * Retourne une représentation textuelle du serveur.
     *
     *
     * @return informations du serveur
     */
    @Override
    public String toString() {
        return "Serveur{" +
                "nom='" + nom + '\'' +
                ", matricule='" + matricule + '\'' +
                '}';
    }
}
